"""Read-only full-matrix audit; export bounded review artifacts after completion."""
from collections import Counter
from pathlib import Path
import hashlib
import json
import subprocess
import xml.etree.ElementTree as ET

from app.evaluation.historical import summarize, verify
from app.tools.repository import is_test_file

Q = Path('E:/multi-agent/.repopilot/quality-v2-20261020')
H = Path('E:/multi-agent/.repopilot/history-20261019')
W = Path('C:/Users/14659/.codex/worktrees/ci-fixtures-model-scope/multi-agent')
D = W / 'docs/repair-quality-v2-20261020'


def sha(p):
    return hashlib.sha256(p.read_bytes()).hexdigest()


def read(p):
    return json.loads(p.read_text(encoding='utf-8'))


def save(name, data):
    (D / name).write_bytes((json.dumps(data, ensure_ascii=False, indent=2) + '\n').encode())


def result_summary(result):
    return {k: result.get(k) for k in ['status', 'passed', 'failed', 'errors', 'skipped', 'evidence', 'image_id']} if result else None


if __name__ == '__main__':
    p = read(Q / 'preregistration.json')
    assert sha(Q / 'preregistration.json') == (Q / 'preregistration.sha256').read_text().strip()
    assert (D / 'preregistration.json').read_bytes() == (Q / 'preregistration.json').read_bytes()
    lock = verify(Path(p['corpus']), p['lock_sha256'])
    assert sha(Q / 'results/controls/controls.json') == p['controls_sha256']
    records = read(Q / 'results/records.json')
    resolved = read(Q / 'results/formal/report.json')
    assert resolved['model_name'] == p['model_name'] and resolved['model_endpoint_host'] == 'muyuan.do'
    for name in ['context_chars', 'max_retries', 'retry_workspace_policy', 'temperature', 'model_timeout', 'model_min_interval', 'test_timeout']:
        assert resolved['configuration'][name] == p['profile'][name]
    assert resolved['configuration']['workers'] == p['workers']
    expected = Counter((c, m, v, n) for c in p['case_order'] for m in p['modes'] for v in p['variants'] for n in range(1))
    assert Counter((r['case_id'], r['mode'], r['variant'], r['variant_repeat']) for r in records) == expected
    assert len(records) == p['planned_tasks'] == 36
    for index, case_id in enumerate(p['case_order']):
        order = ['legacy', 'quality'] if index % 2 == 0 else ['quality', 'legacy']
        expected_order = [(arm, block, mode) for block, arm in enumerate(order)
                          for mode in p['modes'][block:] + p['modes'][:block]]
        assert [(r['variant'], r['repeat'], r['mode']) for r in records if r['case_id'] == case_id] == expected_order
    subprocess.run(['git', 'diff', '--exit-code', p['runtime_commit'], '--', 'app', 'tests', 'frontend', 'scripts', 'sandbox', 'pyproject.toml', '.env.example'], cwd=W, check=True, capture_output=True)
    old_plan_sha = sha(H / 'preregistration.json')
    assert old_plan_sha == '2a1d0034f196b5389ee3d66ae2221edc820ed850760180cc6d933c7d68e8e247'
    assert len(read(H / 'results-02/records.json')) == 24
    assert sum(r['strict_success'] for r in read(H / 'results-02/records.json')) == 17
    prior = read(Path('E:/multi-agent/.repopilot/quality-20261020/results/records.json'))
    assert len(prior) == 72
    assert {v: sum(r['strict_success'] for r in prior if r['variant'] == v) for v in ['legacy', 'quality']} == {'legacy': 26, 'quality': 21}
    assert sha(Path('E:/multi-agent/.repopilot/quality-20261020/results/records.json')) == p['prior_results_preserved']['quality_v1']['records_sha256']
    runs, patches, events, reviews, quality = [], [], [], [], []
    artifacts = ['preregistration.json', 'wheel-artifact.json', 'wheel-full-final.xml',
                 'formal-timing.json', 'results/report.json',
                 'results/records.json', 'results/formal/runs.jsonl', 'results/controls/controls.json', 'results/probe-records.json', 'scope-note.json']
    all_original, created, regression_counts = True, 0, Counter()
    for r in records:
        fields = ['case_id', 'variant', 'variant_repeat', 'repeat', 'mode', 'task_id', 'execution_status', 'error',
                  'model_calls', 'tokens', 'duration_seconds', 'retry_count', 'attempts', 'candidate_number',
                  'strict_success', 'original_tests_unchanged', 'diff_digest', 'budget', 'budget_failure', 'provider_failure',
                  'patch_rejections', 'creation_failures', 'review_decisions']
        item = {k: r.get(k) for k in fields}
        item['cohort'] = next(k for k, ids in p['cohorts'].items() if r['case_id'] in ids)
        item['tests'] = {k: result_summary(r.get(k)) for k in ['baseline', 'test_result', 'acceptance']}
        item['failure_observations'] = [{'attempt': f['attempt'], 'observations': [{k: o.get(k) for k in ['file', 'test', 'origin', 'message']} for o in f.get('observations', [])]} for f in r.get('repair_feedback', [])]
        if r['task_id'] is None:
            assert not r['strict_success'] and r['execution_status'] in {'blocked', 'invalid_case'}
            item.update(current_regression=None, recorded_tokens_with_partial=0)

        created += 1

        state = read(path)

        assert Path(state['task_path']).resolve().is_relative_to((Q / 'data' / r['variant']).resolve())
        assert state.get('repair_quality_version') == (2 if r['variant'] == 'quality' else None)
        assert state['approval_status'] == 'pending' and state['budget_limits'] == p['budget_limits']
        assert r['budget']['pending_reservations'] == 0
        assert r['budget']['model_requests'] == r['model_calls'] <= p['budget_limits']['model_requests']
        model_events = [e for e in state['execution_history'] if e['node'] == 'model']

        observed = sum(e.get('input_tokens', 0) + e.get('output_tokens', 0) for e in model_events)
        assert observed == r['budget']['observed_tokens']
        if r['tokens'] is not None:
            assert observed == r['tokens']
        item['recorded_tokens_with_partial'] = observed

        workspace = Path(state['workspace_path'])
        for n in range(1, state['candidate_number'] + 1):
            candidate = Path(state['task_path']) / f'attempts/{n:04}/workspace'
            assert all(sha(candidate / name) == fingerprint for name, fingerprint in case['source_manifest'].items() if name.endswith('.py') and is_test_file(name))
        assert r['original_tests_unchanged']
        current_witnesses = [w for w in state.get('regression_results', []) if w['attempt'] == state['attempts'] and w['diff_digest'] == state['diff_digest'] and w.get('candidate_id') == state.get('candidate_id')]
        witness = current_witnesses[-1] if current_witnesses else None

        if witness:

        strict = r['baseline_verified'] and r['execution_status'] == 'awaiting_approval' and (r.get('test_result') or {}).get('status') == (r.get('acceptance') or {}).get('status') == 'passed'
        assert bool(strict) == r['strict_success']

        for test in [r['baseline'], *state['test_results'], r.get('acceptance'), *[w.get('baseline_result') for w in state.get('regression_results', [])]]:

        quality_events = [e for e in state['execution_history'] if e['node'] in {'candidate_quality', 'review_evidence', 'contract_api_context'}]
        for e in quality_events:
            root = Path(state['task_path']) / f"attempts/{e['candidate_number']:04}/workspace"
            if e['node'] == 'candidate_quality':
                assert all(sha(root / name) == fingerprint for name, fingerprint in e['staged_content_digests'].items())
                raw = json.dumps(e['raw_proposal'], sort_keys=True, separators=(',', ':'), ensure_ascii=False)
                assert hashlib.sha256(raw.encode()).hexdigest() == e['raw_proposal_digest']
                assert e['version'] == 2
            elif e['node'] == 'contract_api_context':
                baseline = root.parent / 'baseline'

                assert all(hashlib.sha256((baseline / f['path']).read_text(encoding='utf-8').encode()).hexdigest() == f['source_text_sha256'] for f in e['api_slices'])
            else:
                assert all(hashlib.sha256((root / f['path']).read_text(encoding='utf-8').encode()).hexdigest() == f['source_text_sha256'] for f in e['file_facts'])

        if strict and r['variant'] == 'quality':

            artifacts.append(candidate.relative_to(Q).as_posix())
        patches.append({'task_id': r['task_id'], 'case_id': r['case_id'], 'variant': r['variant'], 'mode': r['mode'], 'repeat': r['variant_repeat'], 'attempts': attempts})
        artifacts.append(path.relative_to(Q).as_posix())
        runs.append(item)
    for f in p['app_files']:
        assert sha(W / f['path']) == f['sha256']
        assert sha(Path('E:/multi-agent/.repopilot/ci-isolation-venv/Lib/site-packages') / f['path']) == f['sha256']
    previous_tests = subprocess.check_output(['git', 'ls-tree', '-r', '--name-only', '05c00e0adbd778a8257ed886cf5501a2ca9b24a0', '--', 'tests'], cwd=W, text=True).splitlines()
    assert len(previous_tests) == 78
    for name in previous_tests:

        assert before.replace(b'\r\n', b'\n') == (W / name).read_bytes().replace(b'\r\n', b'\n')
    suites = list(ET.parse(Q / 'wheel-full-final.xml').getroot().iter('testsuite'))

    assert sum(int(s.get('failures', 0)) + int(s.get('errors', 0)) for s in suites) == 0

    groups = {group: {variant: summarize([r for r in records if r['case_id'] in ids and r['variant'] == variant]) for variant in p['variants']} for group, ids in cohorts.items()}
    for group, ids in cohorts.items():
        for variant, modes in groups[group].items():
            for mode, metrics in modes.items():

                assert metrics['planned_runs'] == len(ids) == len(subset)

    controls = read(Q / 'results/controls/controls.json')

    print(json.dumps({'audit_passed': True, 'groups': groups, 'statuses': report['execution_statuses']}, ensure_ascii=False))
