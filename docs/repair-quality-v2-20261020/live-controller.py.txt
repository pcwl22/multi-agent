"""Memory-only credential; finite frozen calibration; no external writes."""
import getpass
import hashlib
import json
import logging
import os
import subprocess
import time
from datetime import datetime, timezone
from pathlib import Path
import xml.etree.ElementTree as ET
import httpx
from pydantic import SecretStr
from app.config.settings import Settings
from app.evaluation.historical import run
from app.models.budgets import BudgetLimits

Q = Path('E:/multi-agent/.repopilot/quality-v2-20261020')
W = Path('C:/Users/14659/.codex/worktrees/ci-fixtures-model-scope/multi-agent')
os.environ['HTTPS_PROXY'] = 'http://127.0.0.1:7890'
logging.getLogger('repopilot.graph').setLevel(logging.CRITICAL)
secret = getpass.getpass('Provider key (hidden): ')
try:
    assert secret
    print('CREDENTIAL_READY; commands RUN, CLOSE', flush=True)
    command = input().strip()
    if command == 'RUN':
        raw = (Q / 'preregistration.json').read_bytes()
        assert hashlib.sha256(raw).hexdigest() == (Q / 'preregistration.sha256').read_text().strip()
        p = json.loads(raw)
        assert p['planned_tasks'] == 36
        assert hashlib.sha256((Q / 'results/controls/controls.json').read_bytes()).hexdigest() == p['controls_sha256']
        artifact = json.loads((Q / 'wheel-artifact.json').read_text(encoding='utf-8'))
        assert hashlib.sha256(Path(artifact['path']).read_bytes()).hexdigest() == p['wheel_sha256']
        import app
        for row in p['app_files']:
            assert hashlib.sha256((W / row['path']).read_bytes()).hexdigest() == row['sha256']
            assert hashlib.sha256((Path(app.__file__).parent.parent / row['path']).read_bytes()).hexdigest() == row['sha256']
        subprocess.run(['git', 'diff', '--exit-code', p['runtime_commit'], '--', 'app', 'tests', 'frontend', 'scripts', 'sandbox', 'pyproject.toml', '.env.example'], cwd=W, check=True, capture_output=True)
        assert hashlib.sha256((Q / 'wheel-full-final.xml').read_bytes()).hexdigest() == p['junit_sha256']
        suites = list(ET.parse(Q / 'wheel-full-final.xml').getroot().iter('testsuite'))
        assert sum(int(s.get('failures', 0)) + int(s.get('errors', 0)) for s in suites) == 0
        assert sum(int(s.get('tests', 0)) for s in suites) == p['verified_test_count']
        response = httpx.get(p['model_base_url'] + '/models', headers={'Authorization': 'Bearer ' + secret}, timeout=30)
        response.raise_for_status()
        assert p['model_name'] in {x['id'] for x in response.json()['data']}
        del response
        base = Settings(_env_file=None, repository_root=Path(p['corpus']) / 'sources', data_dir=Q / 'data/unused',
            model_mode='live', model_name=p['model_name'], model_base_url=p['model_base_url'], openai_api_key=SecretStr(secret),
            model_streaming=True, model_stream_usage=True, model_timeout=120, model_min_interval=7.5, temperature=0,
            context_chars=24000, model_path_scope=True, repair_quality_checks=False, repair_quality_policy_version=2,
            sandbox_image=p['image'], test_timeout=120, max_retries=1, retry_workspace_policy='isolated',
            patch_transactions=True, task_budgets=True, budget_limits=BudgetLimits(**p['budget_limits']))
        variants = {name: base.model_copy(update={'data_dir': Q / 'data' / name, 'repair_quality_checks': name == 'quality'}) for name in ['legacy', 'quality']}
        timing = {'started_at': datetime.now(timezone.utc).isoformat(), 'preregistration_sha256': hashlib.sha256(raw).hexdigest()}
        def save_timing():
            (Q / 'formal-timing.json').write_text(json.dumps(timing, indent=2) + '\n', encoding='utf-8')
        save_timing()
        print(json.dumps({'formal_start': True, 'planned_tasks': 36, **timing}), flush=True)
        started = time.monotonic()
        result = run(Path(p['corpus']), Q / 'results', p['lock_sha256'], base, repeats=2, workers=2, variants=variants)
        timing.update(completed_at=datetime.now(timezone.utc).isoformat(), wall_seconds=time.monotonic() - started)
        save_timing()
        print(json.dumps({'formal_complete': True, 'runs': result['runs'], 'variants': result['variants']}), flush=True)
except Exception as exc:
    print(json.dumps({'controller_error': type(exc).__name__, 'detail': str(exc).replace(secret, '[REDACTED]')[:1200]}), flush=True)
finally:
    secret = ''
    print('CREDENTIAL_SESSION_CLOSED', flush=True)
