    return hashlib.sha256(p.read_bytes()).hexdigest()
def read(p):
    return json.loads(p.read_text(encoding='utf-8'))
def save(name, data):
    (D / name).write_bytes((json.dumps(data, ensure_ascii=False, indent=2) + '\n').encode())

ledgers, paths = [], []
for path in sorted((Q / 'data').glob('*/budget.sqlite3')):
    with sqlite3.connect(path.as_uri() + '?mode=ro', uri=True) as db:
        binding, requests, tokens, incomplete = db.execute('SELECT binding,model_requests,tokens,incomplete FROM budget').fetchone()
        pending = db.execute('SELECT count(*) FROM reservations WHERE completed=0').fetchone()[0]
    ledgers.append({'id': path.parent.name, 'binding': json.loads(binding), 'model_request_reservations': requests,
        'observed_tokens': tokens, 'usage_incomplete_flag': bool(incomplete), 'pending_reservations': pending})
    paths.append(path)
assert [r['id'] for r in ledgers] == ['probe-01', 'probe-02']
assert [r['model_request_reservations'] for r in ledgers] == [1, 1]
assert [r['pending_reservations'] for r in ledgers] == [0, 1]
