    return hashlib.sha256(p.read_bytes()).hexdigest()
def read(p):
    return json.loads(p.read_text(encoding='utf-8'))

tests['passed'] = tests['tests'] - tests['failures'] - tests['errors'] - tests['skipped']
assert tests == {'tests': 343, 'failures': 0, 'errors': 0, 'skipped': 2, 'passed': 341}, tests
fault_suites = list(ET.parse(Q / 'fault-controls-final.xml').getroot().iter('testsuite'))
